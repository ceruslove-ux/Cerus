    h6 = bit32.band(bit32.bxor(h6, bit32.lrotate(h3, i % 5)) + c, 0xffffffff)
  end
  return string.format("%08x%08x%08x%08x%08x%08x", h1, h2, h3, h4, h5, h6)
end
local _E = _mix(_P)

-- Anti-tamper: payload hash + native stamp presence
local function _check()
  if type(_P) ~= "string" or #_P < 16 then return false end
  if _mix(_P) ~= _E then return false end
  if _P == _P_ALT then return false end
  if type(__NATIVE_STAMP) ~= "string" or #__NATIVE_STAMP < 8 then return false end
  return true
end

local function decode_standard_b64(s)
  local alph = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
  local map = {}
  for i = 1, #alph do map[string.byte(alph, i)] = i - 1 end
  s = tostring(s or ""):gsub("=+$", "")
  local out = {}
  local i, n = 1, #s
  while i <= n do
    local a = map[string.byte(s, i)] or 0
    local b = map[string.byte(s, i + 1)] or 0
    local c = map[string.byte(s, i + 2)] or 0
    local d = map[string.byte(s, i + 3)] or 0
    local v = a * 262144 + b * 4096 + c * 64 + d
    out[#out + 1] = string.char(math.floor(v / 65536) % 256)
    if i + 2 <= n then out[#out + 1] = string.char(math.floor(v / 256) % 256) end
    if i + 3 <= n then out[#out + 1] = string.char(v % 256) end
    i = i + 4
  end
  return table.concat(out)
end

local function xor_stream(data, key, salt)
  if type(data) ~= "string" or type(key) ~= "string" or #key < 1 then return "" end
  local o = {}
  local klen = #key
  salt = salt or 0
  for i = 1, #data do
    local kb = string.byte(key, ((i - 1) % klen) + 1)
    local roll = bit32.band(i * 0x9e + bit32.rshift(i, 3) + salt, 0xff)
    o[i] = string.char(bit32.bxor(string.byte(data, i), kb, roll))
  end
  return table.concat(o)
end
local function dual_xor(data)
  return xor_stream(xor_stream(data, _k1(), 0x61), _k2(), 0x25)
end

-- Native++ reverse (C++! header + CRC + rotate undo)
local function native_unwrap(blob)
  if type(blob) ~= "string" or #blob < 9 then return nil end
  if string.byte(blob, 1) ~= 0x43 or string.byte(blob, 2) ~= 0x2b
     or string.byte(blob, 3) ~= 0x2b or string.byte(blob, 4) ~= 0x21 then
    return nil
  end
  local crc = string.byte(blob, 5) + string.byte(blob, 6) * 256
      + string.byte(blob, 7) * 65536 + string.byte(blob, 8) * 16777216
  local k0 = bit32.bxor(crc, 0xC3C3C3C3)
  local body = string.sub(blob, 9)
  local o = {}
  local rots = {3, 5, 7, 11}
  for i = 1, #body do
    local rot = rots[((i - 1) % 4) + 1]
    local v = string.byte(body, i)
    -- ror
    v = bit32.band(bit32.bor(bit32.rshift(v, rot), bit32.lshift(v, 8 - rot)), 0xff)
    v = bit32.bxor(v, bit32.band(bit32.rshift(k0, bit32.band(i * 7, 24)), 0xff), bit32.band(i * 0x9e, 0xff))
    o[i] = string.char(v)
  end
  return table.concat(o)
end

local function decode_b64_strong(s)
  if not _check() then return nil end
  local function codepoints(str)
    local t = {}
    local i = 1
    local n = #str
    while i <= n do
      local c = string.byte(str, i)
      if not c then break end
      if c < 0x80 then t[#t + 1] = c; i = i + 1
      elseif c < 0xE0 then
        local b2 = string.byte(str, i + 1) or 0
        t[#t + 1] = c * 64 + b2 - 0x3080; i = i + 2
      elseif c < 0xF0 then
        local b2, b3 = string.byte(str, i + 1) or 0, string.byte(str, i + 2) or 0
        t[#t + 1] = ((c % 16) * 4096) + ((b2 % 64) * 64) + (b3 % 64); i = i + 3
      else i = i + 4; t[#t + 1] = 0 end
    end
    return t
  end
  local alphCP = { codepoints(_A), codepoints(_A2) }
  local maps = {}
  for mi = 1, 2 do
    local m = {}
    for i, cp in ipairs(alphCP[mi]) do m[cp] = i - 1 end
    maps[mi] = m
  end
  local cps = codepoints((s:gsub("=+$", "")))
  local out = {}
  local blk, i, n = 0, 1, #cps
  while i <= n do
    local map = maps[(blk % 2) + 1]
    local a = map[cps[i]] or 0
    local b = map[cps[i + 1]] or 0
    local c = map[cps[i + 2]] or 0
    local d = map[cps[i + 3]] or 0
    a = bit32.band(bit32.bxor(a, 0x05) - (blk % 8), 63)
    b = bit32.band(b - blk * 7 - 0x11, 63)
    c = bit32.band(c - blk * 13 - 0x25, 63)
    d = bit32.band(bit32.bxor(d, 0x03) - bit32.rshift(blk, 1) % 16, 63)
    local v = a * 262144 + b * 4096 + c * 64 + d
    out[#out + 1] = string.char(math.floor(v / 65536) % 256)
    if i + 2 <= n then out[#out + 1] = string.char(math.floor(v / 256) % 256) end
    if i + 3 <= n then out[#out + 1] = string.char(v % 256) end
    i = i + 4; blk = blk + 1
  end
  return table.concat(out)
end

local decode_payload = decode_b64_strong
local decode_primary = decode_standard_b64
local decode_vm_blob = decode_b64_strong

local function Buffer(raw)
  raw = tostring(raw or "")
  local self = { _b = raw, _i = 1, _n = #raw }
  function self:eof() return self._i > self._n end
  function self:need(k) return (self._i + k - 1) <= self._n end
  function self:u8()
    if not self:need(1) then return nil end
    local v = string.byte(self._b, self._i); self._i = self._i + 1; return v
  end
  function self:u16()
    if not self:need(2) then return nil end
    local a, b = string.byte(self._b, self._i), string.byte(self._b, self._i + 1)
    self._i = self._i + 2; return a + b * 256
  end
  function self:i16()
    local v = self:u16(); if v == nil then return nil end
    if v >= 0x8000 then v = v - 0x10000 end; return v
  end
  function self:str(n)
    n = tonumber(n) or 0
    if n < 0 or not self:need(n) then return nil end
    local s = string.sub(self._b, self._i, self._i + n - 1)
    self._i = self._i + n; return s
  end
  function self:seek(rel)
    local ni = self._i + (tonumber(rel) or 0)
    if ni < 1 or ni > self._n + 1 then return false end
    self._i = ni; return true
  end
  return self
end

local function exec_vm(env, bc)
  if type(bc) ~= "string" or #bc < 2 then return nil end
  if not _check() then return nil end
  local buf = Buffer(bc)
  local stack = {}
  local function push(v) stack[#stack + 1] = v end
  local function pop()
    local n = #stack; if n < 1 then return nil end
    local v = stack[n]; stack[n] = nil; return v
  end
  local op = {}
  op[0x11] = function()
    local n = buf:u16(); if not n then return "fail" end
    local s = buf:str(n); if s == nil then return "fail" end; push(s)
  end
  op[0x12] = function() local v = buf:u8(); if v == nil then return "fail" end; push(v) end
  op[0x13] = function() local v = buf:i16(); if v == nil then return "fail" end; push(v) end
  op[0x22] = function()
    local name = pop()
    local v = env[name]
    if v == nil and type(_G) == "table" then v = rawget(_G, name) end
    if name == "loadstring" or name == "load" then v = env.loadstring or env.load or v end
    push(v)
  end
  op[0x23] = function() local key = pop(); local tab = pop(); push(tab and tab[key]) end
  op[0x24] = function() local val = pop(); local key = pop(); local tab = pop(); if type(tab) == "table" then tab[key] = val end end
  op[0x31] = function() local b = pop(); local a = pop(); push(tostring(a or "") .. tostring(b or "")) end
  op[0x32] = function()
    local data = pop(); local key = pop()
    if type(data) ~= "string" or type(key) ~= "string" or #key < 1 then return "fail" end
    local o = {}
    for i = 1, #data do
      local kb = string.byte(key, ((i - 1) % #key) + 1)
      o[i] = string.char(bit32.bxor(string.byte(data, i), kb))
    end
    push(table.concat(o))
  end
  op[0x33] = function()
    local argc = buf:u8(); if argc == nil then return "fail" end
    local args = {}
    for i = argc, 1, -1 do args[i] = pop() end
    local fn = pop()
    if type(fn) ~= "function" then return "fail" end
    local ok, res = pcall(fn, unpack(args, 1, argc))
    if not ok then return "fail" end
    push(res)
  end
  op[0x34] = op[0x33]
  op[0x41] = function() local off = buf:i16(); if off == nil or not buf:seek(off) then return "fail" end end
  op[0x42] = function() local off = buf:i16(); if off == nil then return "fail" end; local v = pop(); if not v then if not buf:seek(off) then return "fail" end end end
  op[0x43] = function() local off = buf:i16(); if off == nil then return "fail" end; local v = pop(); if v then if not buf:seek(off) then return "fail" end end end
  op[0x44] = function() return "ret" end
  op[0x51] = function() push(stack[#stack]) end
  op[0x52] = function() local a = pop(); local b = pop(); push(a); push(b) end
  op[0x53] = function() pop() end
  op[0x54] = function() local a = pop(); local b = pop(); local c = pop(); push(b); push(a); push(c) end
  op[0x61] = function() local b = pop(); local a = pop(); push((tonumber(a) or 0) + (tonumber(b) or 0)) end
  op[0x62] = function() local b = pop(); local a = pop(); push((tonumber(a) or 0) - (tonumber(b) or 0)) end
  op[0x63] = function() local b = pop(); local a = pop(); push((tonumber(a) or 0) * (tonumber(b) or 0)) end
  op[0x64] = function() local b = pop(); local a = pop(); push(bit32.band(tonumber(a) or 0, tonumber(b) or 0)) end
  op[0x65] = function() local b = pop(); local a = pop(); push(bit32.bxor(tonumber(a) or 0, tonumber(b) or 0)) end
  op[0x77] = function() local _ = buf:u8() end
  op[0x88] = function() if not _check() then return "fail" end end
  op[0xFE] = function() _trap(); return "fail" end

  local steps = 0
  while not buf:eof() do
    steps = steps + 1
    if steps > 5000000 then return nil end
    if steps % 512 == 0 and not _check() then return nil end
    local code = buf:u8()
    if code == nil then return nil end
    local handler = op[code]
    if not handler then return nil end
    local st = handler()
    if st == "ret" then return pop() end
    if st == "fail" then return nil end
  end
  return pop()
end

if not _check() then return _abort() end

-- Pipeline: strong b64 -> dual xor -> native++ unwrap -> outer VM
local outer_enc = decode_vm_blob(_P)
if type(outer_enc) ~= "string" then return _abort() end
local outer_x = dual_xor(outer_enc)
local outer_bin = native_unwrap(outer_x)
if type(outer_bin) ~= "string" then
  -- fallback if stamp path fails: treat as plain dual-xor body
  outer_bin = outer_x
end
if type(outer_bin) ~= "string" or #outer_bin < 2 then return _abort() end

local LS = loadstring or load
local LD = load or loadstring
local nest_token = exec_vm({ print = print, warn = warn, loadstring = LS, load = LD }, outer_bin)
if type(nest_token) ~= "string" or #nest_token < 4 then return _abort() end

local inner_x = decode_standard_b64(nest_token)
local inner_bytes = dual_xor(inner_x)
if type(inner_bytes) ~= "string" or #inner_bytes < 2 then return _abort() end

local ok = pcall(function()
  exec_vm({ print = print, warn = warn, loadstring = LS, load = LD }, inner_bytes)
end)
if not ok then return _abort() end
